cmake_minimum_required(VERSION 3.0.2)
project(atr_imu_ros)
find_package(catkin REQUIRED COMPONENTS rospy sensor_msgs diagnostic_msgs)
catkin_python_setup()
catkin_package()
catkin_install_python(PROGRAMS scripts/imu_node DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_protocol.py)
  catkin_add_nosetests(test/test_messages.py)
endif()
